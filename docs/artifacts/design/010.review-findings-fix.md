```go
func (w *WebhookNotifier) Send(notifications []*Notification) error {
	if len(notifications) == 0 {
		return nil
	}
	if w.client == nil {
		w.client = &slackClient{webhookURL: w.webhookURL}
	}

	grouped := groupNotificationsByStatus(notifications)
	var text string

	for _, status := range summaryStatusOrder {
		notifs, ok := grouped[status]
		if !ok {
			continue
		}
		text += statusMessages[status] + "\n"
		for _, n := range notifs {
			text += formatNotificationLine(n) + "\n"
			if len(n.Targets) > 0 {
				text += "→ " + formatMentions(n.Targets) + "\n"
			}
		}
		text += "\n"
	}

	return w.client.PostWebhook(&slack.WebhookMessage{Text: text})
}
```

```go
func (d *DMNotifier) Send(notifications []*Notification) error {
	if len(notifications) == 0 {
		return nil
	}
	if d.client == nil {
		d.client = slack.New(d.token)
	}

	byTarget := groupNotificationsByTarget(notifications)

	// 한 대상이 실패해도 나머지 전송은 계속하되, 실패는 반드시 반환한다.

	// 중복제거에 걸려 영구 유실된다.
	var failed []string
	for slackID, notifs := range byTarget {
		text := "당신에게 액션이 필요한 MR이 있습니다:\n\n"
		for _, n := range notifs {
			text += fmt.Sprintf("%s\n• %s\n\n",
				statusMessages[n.MR.Status], formatNotificationLine(n))
		}

		if _, _, err := d.client.PostMessage(slackID, slack.MsgOptionText(text, false)); err != nil {
			log.Printf("error sending DM to %s: %v", slackID, err)
			failed = append(failed, slackID)
		}
	}

	if len(failed) > 0 {
		return fmt.Errorf("failed to send DM to %d of %d targets: %v",
			len(failed), len(byTarget), failed)
	}
	return nil
}
```

```go
func (w *WebhookNotifier) Send(notifications []*Notification) error {
	if len(notifications) == 0 {
		return nil
	}
	if w.client == nil {
		w.client = &slackClient{webhookURL: w.webhookURL}
	}

	messages := buildWebhookMessages(notifications)
	for i, msg := range messages {
		if err := w.client.PostWebhook(&slack.WebhookMessage{Text: msg}); err != nil {
			return fmt.Errorf("error sending notification message %d/%d: %w", i+1, len(messages), err)
		}
	}
	return nil
}

// buildWebhookMessages groups notifications by status and splits them into
// chunks of at most maxMRsPerMessage MRs, mirroring the summary path.
// Slack rejects webhook payloads beyond roughly 40,000 characters; on the
// first run every open MR counts as changed, so a single message overflows.
func buildWebhookMessages(notifications []*Notification) []string {
	grouped := groupNotificationsByStatus(notifications)

	var messages []string
	var current string
	var count int

	flush := func() {
		if current != "" {
			messages = append(messages, current)
			current = ""
			count = 0
		}
	}

	for _, status := range summaryStatusOrder {
		notifs, ok := grouped[status]
		if !ok {
			continue
		}

		header := statusMessages[status] + "\n"
		current += header

		for _, n := range notifs {
			if count >= maxMRsPerMessage {
				flush()
				current += header // 이어지는 메시지에도 상태 헤더를 다시 넣는다
			}
			current += formatNotificationLine(n) + "\n"
			if len(n.Targets) > 0 {
				current += "→ " + formatMentions(n.Targets) + "\n"
			}
			count++
		}
		current += "\n"
	}

	flush()
	return messages
}
```

```go
// dedupeIDs removes duplicate IDs while preserving first-seen order.
// A project reachable both through a configured group and through the
// projects list would otherwise be crawled twice, duplicating every
// notification and doubling the approvals API calls.
func dedupeIDs(ids []int) []int {
	seen := make(map[int]struct{}, len(ids))
	result := make([]int, 0, len(ids))
	for _, id := range ids {
		if _, ok := seen[id]; ok {
			continue
		}
		seen[id] = struct{}{}
		result = append(result, id)
	}
	return result
}
```

```go
// saveState writes the notification state atomically: it writes a temporary
// file in the same directory and renames it over the target. A partial write
// (SIGKILL, OOM, disk full) would otherwise leave truncated JSON, which the
// next run treats as "no previous state" and re-notifies every MR.
func saveState(path string, state *NotificationState) error {
	if path == "" {
		return nil
	}

	state.UpdatedAt = time.Now()

	data, err := json.MarshalIndent(state, "", "  ")
	if err != nil {
		return fmt.Errorf("error marshaling state: %w", err)
	}

	dir := filepath.Dir(path)
	tmp, err := os.CreateTemp(dir, ".notification_state-*.tmp")
	if err != nil {
		return fmt.Errorf("error creating temp state file: %w", err)
	}
	tmpName := tmp.Name()
	defer os.Remove(tmpName) // rename에 성공하면 아무것도 지우지 않는다

	if _, err := tmp.Write(data); err != nil {
		tmp.Close()
		return fmt.Errorf("error writing temp state file: %w", err)
	}
	if err := tmp.Sync(); err != nil {
		tmp.Close()
		return fmt.Errorf("error syncing temp state file: %w", err)
	}
	if err := tmp.Close(); err != nil {
		return fmt.Errorf("error closing temp state file: %w", err)
	}
	if err := os.Chmod(tmpName, 0644); err != nil {
		return fmt.Errorf("error setting state file mode: %w", err)
	}
	if err := os.Rename(tmpName, path); err != nil {
		return fmt.Errorf("error replacing state file: %w", err)
	}

	return nil
}
```

```go
// LambdaInput represents the input event for the Lambda function.
// EventBridge rules select which job to run: {"job":"summary"} for the
// periodic report, anything else (or an empty event) for notifications.
type LambdaInput struct {
	Job string `json:"job"`
}

func (i LambdaInput) normalizedJob() string {
	if i.Job == "" {
		return "notify"
	}
	return i.Job
}

// HandleRequest is the main Lambda handler function.
func HandleRequest(ctx context.Context, input LambdaInput) (string, error) {
	config, err := loadConfig(&OsEnv{})
	if err != nil {
		log.Printf("Error loading configuration: %v", err)
		return "", err
	}
	return runLambdaJob(config, input)
}

func runLambdaJob(config *Config, input LambdaInput) (string, error) {
	switch input.normalizedJob() {
	case "notify":
		if err := execute(config); err != nil {
			return "", err
		}
		return "Success", nil
	case "summary":
		if err := executeSummary(config); err != nil {
			return "", err
		}
```
